# Module: xhci.build
# Purpose: Build portable discovery and an opt-in genuine ReactOS diagnostic.
# Created: 2026-10-25
# Architecture: All outputs live in Binaries/.build; external SDK inputs are read-only.
cmake_minimum_required(VERSION 3.19)
project(xhci_reactos LANGUAGES CXX)
option(XHCI_TESTS "Build discovery boundary checks" ON)
option(XHCI_SANITIZERS "Enable host ASan/UBSan" OFF)
function(xhci_cpp target)
    set_target_properties(${target} PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /permissive- /GR- /EHs-c-)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-exceptions -fno-rtti)
    endif()
    if(XHCI_SANITIZERS)
        if(MSVC OR CMAKE_CROSSCOMPILING)
            message(FATAL_ERROR "Sanitizers require a native GCC/Clang host")
        endif()
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
endfunction()
add_library(xhci_core STATIC src/core/controller.cpp src/core/firmware.cpp src/core/resources.cpp)
target_include_directories(xhci_core PUBLIC src/core)
xhci_cpp(xhci_core)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_executable(xhci_inventory src/tools/inventory.cpp)
    target_link_libraries(xhci_inventory PRIVATE xhci_core)
    xhci_cpp(xhci_inventory)
endif()
if(XHCI_TESTS)
    enable_testing()
    add_executable(xhci_core_tests tests/controller_tests.cpp)
    target_link_libraries(xhci_core_tests PRIVATE xhci_core)
    xhci_cpp(xhci_core_tests)
    add_test(NAME xhci_discovery_boundaries COMMAND xhci_core_tests)
endif()
if(XHCI_TESTS)
    add_executable(xhci_firmware_tests tests/firmware_tests.cpp)
    target_link_libraries(xhci_firmware_tests PRIVATE xhci_core)
    xhci_cpp(xhci_firmware_tests)
    add_test(NAME xhci_firmware_handoff COMMAND xhci_firmware_tests)
endif()
if(XHCI_NATIVE_DIAGNOSTIC OR XHCI_NATIVE_CONTROLLER OR XHCI_NATIVE_BOOT OR XHCI_NATIVE_USB_COMPATIBILITY OR XHCI_IMAGE_TOOLS)
    include(cmake/dependencies.cmake)
endif()
include(cmake/m03_m04.cmake OPTIONAL)
include(cmake/m05_m06.cmake)
include(cmake/m07_m08.cmake)
include(cmake/native.cmake)
include(cmake/m09_m10.cmake)
